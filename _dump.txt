 479|   function extractDriveFolderId(link) {
 480|     const text = String(link || "").trim();
 481|     const m = text.match(/folders\/([A-Za-z0-9_-]+)/) || text.match(/[?&]id=([A-Za-z0-9_-]+)/);
 482|     if (m) return m[1];
 483|     if (/^[A-Za-z0-9_-]{10,}$/.test(text)) return text;
 484|     return null;
 485|   }

 505|   async function listDriveImages(folderId) {
 506|     const key = driveApiKeyInput.value.trim();
 507|     try { localStorage.setItem("ecowheel-gdrive-apikey", key); } catch { /* noop */ }
 508|     const query = encodeURIComponent(`'${folderId}' in parents and trashed=false and (mimeType contains 'image/')`);
 509|     const fields = encodeURIComponent("nextPageToken,files(id,name,mimeType)");
 510|     let url = `https://www.googleapis.com/drive/v3/files?q=${query}&fields=${fields}&pageSize=250&orderBy=name_natural`;
 511|     const headers = {};
 512|     if (driveAccessToken) headers.Authorization = `Bearer ${driveAccessToken}`;
 513|     else if (key) url += `&key=${encodeURIComponent(key)}`;
 514|     else {
 515|       driveStatus.textContent = "Esta carpeta necesita autorización: pulsa «Autorizar con Google» o pega una clave API.";
 516|       return [];
 517|     }
 518|     const out = [];
 519|     let pageToken = "";
 520|     for (let page = 0; page < 4 && out.length < MAX_BULK_ROUNDS; page += 1) {
 521|       const res = await fetch(url + (pageToken ? `&pageToken=${encodeURIComponent(pageToken)}` : ""), { headers });
 522|       if (res.status === 401 || res.status === 403) {
 523|         driveStatus.textContent = "Drive denegó el acceso (401/403): autoriza con Google o haz la carpeta pública (Lector).";
 524|         return [];
 525|       }
 526|       if (!res.ok) {
 527|         driveStatus.textContent = `Drive respondió ${res.status}. Revisa el enlace o la clave API.`;
 528|         return [];
 529|       }
 530|       const data = await res.json();
 531|       (data.files || []).forEach((f) => {
 532|         if (out.length < MAX_BULK_ROUNDS && f?.id && f?.name) out.push({ id: f.id, name: f.name, mimeType: f.mimeType || "image/jpeg" });
 533|       });
 534|       pageToken = data.nextPageToken || "";
 535|       if (!pageToken) break;
 536|     }
 537|     return out;
 538|   }

 540|   async function downloadDriveBlob(file) {
 541|     if (driveAccessToken) {
 542|       const res = await fetch(`https://www.googleapis.com/drive/v3/files/${encodeURIComponent(file.id)}?alt=media`, {
 543|         headers: { Authorization: `Bearer ${driveAccessToken}` },
 544|       });
 545|       if (!res.ok) throw new Error(`Drive media ${res.status}`);
 546|       return await res.blob();
 547|     }
 548|     const key = driveApiKeyInput.value.trim();
 549|     const pub = `https://drive.google.com/uc?export=download&id=${encodeURIComponent(file.id)}${key ? `&key=${encodeURIComponent(key)}` : ""}`;
 550|     const res = await fetch(pub);
 551|     if (!res.ok) throw new Error(`Drive uc ${res.status}`);
 552|     return await res.blob();
